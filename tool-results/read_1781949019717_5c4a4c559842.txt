   148→def quarter_for(d: date) -> str:
   149→    """Q1 = Apr-Jun, Q2 = Jul-Sep, Q3 = Oct-Dec, Q4 = Jan-Mar."""
   150→    m = d.month

   152→        return "Q1"
   153→    if m in (7, 8, 9):
   154→        return "Q2"

   156→        return "Q3"

   466→            "hcp_id": f"HCP-{i:04d}",
   467→            "first_name": first,
   468→            "last_name": last,
   469→            "specialty": rng.choices(specs, weights=spec_w, k=1)[0],
   470→            "tier": tier,
   471→            "decile": decile,
   472→            "city": city,
   473→            "state": state,
   474→            "zone": zone,
   475→            "hospital": hospital,
   476→            "years_practicing": rng.randint(2, 40),
   477→            "preferred_contact": rng.choices(contacts, weights=[0.4, 0.3, 0.2, 0.1], k=1)[0],
   478→            "npi_like_id": npi_like_id,
   479→        })
   480→    return hcps

   490→        out.append({
   491→            "product_id": pid,
   492→            "product_name": name,
   493→            "molecule": mol,
   494→            "therapy_area": ta,
   495→            "launch_date": iso(launch),
   496→            "mrp_inr": mrp,

   504→# --------------------------------------------------------------------------- #
   505→# 4. VISITS                                                                    #
   506→# --------------------------------------------------------------------------- #
   507→def gen_visits(reps: List[Dict], hcps: List[Dict]) -> List[Dict]:
   508→    """Generate ~60K visits.
   509→
   510→    Each active Field_Rep executes 8-12 visits per working day.
   511→    HCP selection biased: Tier-A 3x/mo, Tier-B 1.5x/mo, Tier-C 0.5x/mo.
   512→    Visit type mix: 70% F2F, 15% Virtual, 10% Group_Detailing, 5% Conference.
   513→    Outcome: 60% Detailed, 25% Briefed, 10% No_Show, 5% Refused.
   514→    Samples scale by tier.
   515→    """
   516→    rng = random.Random(SEED + 2)
   517→    visits: List[Dict] = []
   518→
   519→    # Bucket HCPs by tier for biased selection
   520→    tier_a = [h["hcp_id"] for h in hcps if h["tier"] == "A"]
   521→    tier_b = [h["hcp_id"] for h in hcps if h["tier"] == "B"]
   522→    tier_c = [h["hcp_id"] for h in hcps if h["tier"] == "C"]
   523→    tier_lookup = {h["hcp_id"]: h for h in hcps}
   524→
   525→    # Reps: only Field_Reps make visits
   526→    # For each date, find Field_Reps active on that date (hire_date <= d <= exit_date or Active)
   527→    fr_reps = [r for r in reps if r["role"] == "Field_Rep"]
   528→
   529→    # Build a quick lookup: rep_id -> (hire_date, exit_date or None)
   530→    rep_active_window = {}
   531→    for r in reps:
   532→        exit_dt = None
   533→        if r["exit_date"]:
   534→            exit_dt = date.fromisoformat(r["exit_date"])
   535→        rep_active_window[r["rep_id"]] = (
   536→            date.fromisoformat(r["hire_date"]),
   537→            exit_dt,
   538→        )
   539→
   540→    visit_type_choices = VISIT_TYPES
   541→    visit_type_weights = [0.70, 0.15, 0.10, 0.05]
   542→    outcome_choices = OUTCOMES
   543→    outcome_weights = [0.60, 0.25, 0.10, 0.05]
   544→
   545→    # Map reps to their zone therapy focus (we'll use all 12 products, biased by zone specialty)
   546→    products_by_ta: Dict[str, List[str]] = {"Cardio": [], "Diabetes": [], "GI": [], "Respiratory": []}
   547→    for p in PRODUCTS_DEF:
   548→        products_by_ta[p[3]].append(p[0])
   549→    all_product_ids = [p[0] for p in PRODUCTS_DEF]
   550→    product_lookup = {p[0]: p for p in PRODUCTS_DEF}
   551→
   552→    # Visit counter per day (for visit_id uniqueness)
   553→    visit_counter: Dict[date, int] = {}
   554→
   555→    # Pre-build HCP sampling buckets per tier (we'll pick by tier then random)
   556→    # For biased selection we sample 60% Tier A, 30% Tier B, 10% Tier C
   557→    def pick_hcp() -> str:
   558→        bucket = rng.choices([tier_a, tier_b, tier_c], weights=[0.60, 0.30, 0.10], k=1)[0]
   559→        if not bucket:
   560→            # fallback
   561→            bucket = tier_b or tier_a or tier_c
   562→        return rng.choice(bucket)
   563→
   564→    for d in daterange(START_DATE, END_DATE):
   565→        if not is_working_day(d):
   566→            continue
   567→        # Determine which Field_Reps are active on this date
   568→        active_reps = []
   569→        for r in fr_reps:
   570→            hire, exit_dt = rep_active_window[r["rep_id"]]
   571→            if hire <= d and (exit_dt is None or d <= exit_dt):
   572→                active_reps.append(r)
   573→        # Cap visits per day per rep to 8-12 but globally limit to keep ~60K total
   574→        # 36 months * ~313 working days = ~11,268 days; 50 reps active avg ~ 37 reps * ~10 visits = 370/day -> ~41K visits
   575→        # Increase to 12 visits per day on average to reach ~60K
   576→        for r in active_reps:
   577→            n_visits = rng.randint(1, 3)
   578→            for _ in range(n_visits):
   579→                hcp_id = pick_hcp()
   580→                hcp = tier_lookup[hcp_id]
   581→                vtype = rng.choices(visit_type_choices, weights=visit_type_weights, k=1)[0]
   582→                duration = int(np.clip(np.random.normal(18, 6), 5, 60))
   583→                # Products detailed (1-3), biased toward HCP specialty
   584→                n_prod = rng.choices([1, 2, 3], weights=[0.5, 0.35, 0.15], k=1)[0]
   585→                prods = rng.sample(all_product_ids, n_prod)
   586→                # Samples scaled by tier
   587→                if hcp["tier"] == "A":
   588→                    samples = rng.randint(5, 20)
   589→                elif hcp["tier"] == "B":
   590→                    samples = rng.randint(2, 12)
   591→                else:
   592→                    samples = rng.randint(0, 6)
   593→                outcome = rng.choices(outcome_choices, weights=outcome_weights, k=1)[0]
   594→                followup = rng.random() < 0.25
   595→
   596→                visit_counter[d] = visit_counter.get(d, 0) + 1
   597→                visit_id = f"VIS-{d.strftime('%Y%m%d')}-{visit_counter[d]:03d}"
   598→                visits.append({
   599→                    "visit_id": visit_id,
   600→                    "rep_id": r["rep_id"],
   601→                    "hcp_id": hcp_id,
   602→                    "visit_date": iso(d),
   603→                    "visit_type": vtype,
   604→                    "duration_min": duration,
   605→                    "products_detailed": "|".join(prods),
   606→                    "samples_dropped": samples,
   607→                    "outcome": outcome,
   608→                    "followup_required": "true" if followup else "false",
   609→                })
   610→
   611→    return visits
   612→

   614→# --------------------------------------------------------------------------- #
   615→# 5. SALES (derived from visits within 14 days, same HCP+rep)                  #
   616→# --------------------------------------------------------------------------- #
   617→def gen_sales(reps: List[Dict], hcps: List[Dict], products: List[Dict],
   618→              visits: List[Dict]) -> List[Dict]:
   619→    """Generate ~30K sales derived from visits.
   620→
   621→    Per brief 21.3: a sale follows a visit within 14 days for the same HCP+rep.
   622→
   623→    Each Tier-A HCP generates ~2 invoices per month, Tier-B ~1, Tier-C ~0.3.
   624→    Apply: March & Sept 1.3x, May 0.85x, 8% YoY growth baseline.
   625→    New products: 6-month linear ramp from launch_date.
   626→    Discounts uniform 2-12%.
   627→    """
   628→    rng = random.Random(SEED + 3)
   629→
   630→    tier_lookup = {h["hcp_id"]: h for h in hcps}
   631→    # Build product lookup with launch_date parsed as date object
   632→    product_lookup: Dict[str, Tuple] = {}
   633→    for p in products:
   634→        launch = date.fromisoformat(p["launch_date"])
   635→        # Match PRODUCTS_DEF tuple layout: (id, name, mol, ta, launch, mrp, pack, is_new, prio)
   636→        pd_match = next((pp for pp in PRODUCTS_DEF if pp[0] == p["product_id"]), None)
   637→        product_lookup[p["product_id"]] = pd_match or (
   638→            p["product_id"], p["product_name"], p["molecule"], p["therapy_area"],
   639→            launch, p["mrp_inr"], p["pack_size"], p["is_new_launch"] == "true", p["priority"],
   640→        )
   641→
   642→    # Index visits by (rep_id, hcp_id) -> sorted list of visit_dates
   643→    visit_index: Dict[Tuple[str, str], List[date]] = {}
   644→    for v in visits:
   645→        key = (v["rep_id"], v["hcp_id"])
   646→        visit_index.setdefault(key, []).append(date.fromisoformat(v["visit_date"]))
   647→    for k in visit_index:
   648→        visit_index[k].sort()
   649→
   650→    # Determine target monthly invoice count per HCP based on tier
   651→    # Tier A: 2/month, Tier B: 1/month, Tier C: 0.3/month
   652→    def target_invoices_per_month(tier: str) -> float:
   653→        return {"A": 2.0, "B": 1.0, "C": 0.3}[tier]
   654→
   655→    # Generate sales by iterating month by month
   656→    sales: List[Dict] = []
   657→    invoice_counter: Dict[date, int] = {}
   658→
   659→    # Reps that can make sales (Field Reps + replacements) — anyone who has visits
   660→    # IMPORTANT: sort to ensure deterministic iteration (set ordering is non-deterministic
   661→    # due to Python hash randomization).
   662→    reps_with_visits = sorted(set(v["rep_id"] for v in visits))
   663→
   664→    # Build per-HCP monthly sales target counts
   665→    # We pick which HCPs get invoices in each month via Poisson sampling
   666→    months: List[Tuple[int, int, date, date]] = []
   667→    y, m = 2023, 4
   668→    while (y, m) <= (2026, 3):
   669→        first = date(y, m, 1)
   670→        if m == 12:
   671→            last = date(y, 12, 31)
   672→        else:
   673→            last = date(y, m + 1, 1) - timedelta(days=1)
   674→        months.append((y, m, first, last))
   675→        m += 1
   676→        if m > 12:
   677→            m = 1
   678→            y += 1
   679→
   680→    # 8% YoY growth baseline relative to FY24 (year index from 2024)
   681→    # FY24 = year 0, FY25 = year 1 (1.08x), FY26 = year 2 (1.16x)
   682→    def yoy_factor(d: date) -> float:
   683→        fy = fy_for(d)
   684→        return 1.08 ** (fy - 2024)
   685→
   686→    def ramp_factor(product_id: str, d: date) -> float:
   687→        """6-month linear ramp for new products; 1.0 for established."""
   688→        p = product_lookup[product_id]
   689→        launch = p[4]  # date obj
   690→        if not p[7]:  # is_new_launch == False
   691→            return 1.0
   692→        days_since = (d - launch).days
   693→        if days_since < 0:
   694→            return 0.0
   695→        if days_since >= 180:
   696→            return 1.0
   697→        return days_since / 180.0
   698→
   699→    # Distributors
   700→    distributor_pool = [f"DIST-{i:03d}" for i in range(1, 41)]
   701→
   702→    # Choose product mix per HCP based on specialty
   703→    def pick_product_for_hcp(hcp: Dict, d: date) -> str:
   704→        spec = hcp["specialty"]
   705→        # Map specialty to therapy area preferences
   706→        if spec == "Cardiologist":
   707→            ta_weights = {"Cardio": 0.7, "Diabetes": 0.15, "GI": 0.10, "Respiratory": 0.05}
   708→        elif spec == "Diabetologist":
   709→            ta_weights = {"Diabetes": 0.7, "Cardio": 0.15, "GI": 0.10, "Respiratory": 0.05}
   710→        elif spec == "Gastroenterologist":
   711→            ta_weights = {"GI": 0.7, "Cardio": 0.10, "Diabetes": 0.10, "Respiratory": 0.10}
   712→        elif spec == "Pulmonist":
   713→            ta_weights = {"Respiratory": 0.7, "Cardio": 0.10, "Diabetes": 0.10, "GI": 0.10}
   714→        else:
   715→            ta_weights = {"Cardio": 0.30, "Diabetes": 0.30, "GI": 0.20, "Respiratory": 0.20}
   716→        ta = rng.choices(list(ta_weights.keys()), weights=list(ta_weights.values()), k=1)[0]
   717→        candidates = [p["product_id"] for p in products if p["therapy_area"] == ta]
   718→        # Filter out products not yet launched (launch_date is a date object in tuple[4])
   719→        candidates = [pid for pid in candidates if product_lookup[pid][4] <= d]
   720→        if not candidates:
   721→            # fallback: any launched product
   722→            candidates = [p["product_id"] for p in products if date.fromisoformat(p["launch_date"]) <= d]
   723→        return rng.choice(candidates)
   724→
   725→    for (y, m, first, last) in months:
   726→        # Seasonality for this month
   727→        seasonal = SEASONALITY[m]
   728→        # For each HCP, sample number of invoices this month
   729→        for hcp in hcps:
   730→            tier = hcp["tier"]
   731→            base_rate = target_invoices_per_month(tier)
   732→            # Poisson-like sampling
   733→            lam = base_rate * seasonal
   734→            # Apply YoY factor using a mid-month date
   735→            mid = first + timedelta(days=14)
   736→            lam *= yoy_factor(mid)
   737→            n_invoices = rng.choices([0, 1, 2, 3, 4], weights=[
   738→                max(0.0, 1 - lam),
   739→                min(1.0, lam) * 0.55,
   740→                min(1.0, lam) * 0.30,
   741→                min(1.0, lam) * 0.12,
   742→                min(1.0, lam) * 0.03,
   743→            ], k=1)[0]
   744→            if n_invoices == 0:
   745→                continue
   746→
   747→            # Find a rep that has visited this HCP — pick any rep that visited
   748→            # in the last 90 days (within visit window). If no visits, skip.
   749→            candidate_reps: List[str] = []
   750→            for r_id in reps_with_visits:
   751→                key = (r_id, hcp["hcp_id"])
   752→                if key not in visit_index:
   753→                    continue
   754→                # any visit in the last 90 days (relative to mid-month)?
   755→                cutoff_lo = first - timedelta(days=60)
   756→                cutoff_hi = last
   757→                dates_list = visit_index[key]
   758→                # Quick check: any date in [cutoff_lo, cutoff_hi]
   759→                for vd in dates_list:
   760→                    if cutoff_lo <= vd <= cutoff_hi:
   761→                        candidate_reps.append(r_id)
   762→                        break
   763→            if not candidate_reps:
   764→                continue
   765→
   766→            for _ in range(n_invoices):
   767→                rep_id = rng.choice(candidate_reps)
   768→                # Pick a visit date within last 14 days of invoice date
   769→                key = (rep_id, hcp["hcp_id"])
   770→                visits_for_rep_hcp = visit_index[key]
   771→                # Invoice date: random working day in this month
   772→                inv_date = None
   773→                for _try in range(20):
   774→                    cd = first + timedelta(days=rng.randint(0, (last - first).days))
   775→                    if is_working_day(cd):
   776→                        inv_date = cd
   777→                        break
   778→                if inv_date is None:
   779→                    inv_date = first
   780→                # Need a visit within 14 days before inv_date
   781→                visit_match = None
   782→                for vd in visits_for_rep_hcp:
   783→                    delta = (inv_date - vd).days
   784→                    if 0 <= delta <= 14:
   785→                        visit_match = vd
   786→                        break
   787→                if visit_match is None:
   788→                    # Try a visit within 14 days after inv_date as well (some sales lead visits)
   789→                    for vd in visits_for_rep_hcp:
   790→                        delta = (inv_date - vd).days
   791→                        if -14 <= delta < 0:
   792→                            visit_match = vd
   793→                            break
   794→                if visit_match is None:
   795→                    continue  # no causal link — skip
   796→
   797→                # Pick product
   798→                product_id = pick_product_for_hcp(hcp, inv_date)
   799→                ramp = ramp_factor(product_id, inv_date)
   800→                if ramp == 0:
   801→                    continue
   802→                p = product_lookup[product_id]
   803→                mrp = p[5]
   804→                # Unit price ~85% of MRP
   805→                unit_price = int(mrp * 0.85)
   806→                # Qty packs: depends on tier
   807→                if tier == "A":
   808→                    qty = int(np.clip(np.random.normal(60, 20), 10, 200))
   809→                elif tier == "B":
   810→                    qty = int(np.clip(np.random.normal(30, 12), 5, 120))
   811→                else:
   812→                    qty = int(np.clip(np.random.normal(12, 6), 1, 50))
   813→                # Apply ramp factor (new products sell less)
   814→                qty = max(1, int(qty * ramp))
   815→                discount = round(rng.uniform(0.02, 0.12), 2)
   816→                net_value = int(qty * unit_price * (1 - discount))
   817→                channel = rng.choices(CHANNELS, weights=[0.30, 0.30, 0.25, 0.15], k=1)[0]
   818→                distributor = rng.choice(distributor_pool)
   819→
   820→                invoice_counter[inv_date] = invoice_counter.get(inv_date, 0) + 1
   821→                invoice_id = f"INV-{inv_date.strftime('%Y%m%d')}-{invoice_counter[inv_date]:03d}"
   822→                sales.append({
   823→                    "invoice_id": invoice_id,
   824→                    "distributor_id": distributor,
   825→                    "rep_id": rep_id,
   826→                    "hcp_id": hcp["hcp_id"],
   827→                    "product_id": product_id,
   828→                    "qty_packs": qty,
   829→                    "unit_price_inr": unit_price,
   830→                    "discount_pct": f"{discount:.2f}",
   831→                    "net_value_inr": net_value,
   832→                    "invoice_date": iso(inv_date),
   833→                    "channel": channel,
   834→                })
   835→
   836→    return sales
   837→

   840→# 6. EXPENSES                                                                  #
   841→# --------------------------------------------------------------------------- #
   842→def gen_expenses(reps: List[Dict]) -> List[Dict]:
   843→    """Generate ~10K expense entries.
   844→
   845→    Each rep incurs 5-8 expense entries per week.
   846→    Travel 40%, Food 25%, Lodging 15%, Samples 10%, Other 10%.
   847→    Amounts: Travel 200-2000, Food 100-800, Lodging 1500-4000, Samples 50-1500.
   848→    Policy compliant ~95%; reimbursed ~90%.
   849→    """
   850→    rng = random.Random(SEED + 4)
   851→    expenses: List[Dict] = []
   852→
   853→    # Only Field_Reps incur expenses (Area_Managers + above have separate T&E)
   854→    eligible_reps = [r for r in reps if r["role"] == "Field_Rep"]
   855→
   856→    cat_weights = {"Travel": 0.40, "Food": 0.25, "Lodging": 0.15, "Samples": 0.10,
   857→                   "Conference": 0.03, "Mobile": 0.05, "Other": 0.02}
   858→    cat_amounts = {
   859→        "Travel": (200, 2000),
   860→        "Food": (100, 800),
   861→        "Lodging": (1500, 4000),
   862→        "Samples": (50, 1500),
   863→        "Conference": (500, 5000),
   864→        "Mobile": (100, 800),
   865→        "Other": (50, 2000),
   866→    }
   867→    notes_templates = {
   868→        "Travel": ["Cab to hospital", "Auto to clinic", "Fuel reimbursement", "Train ticket", "Bus fare"],
   869→        "Food": ["Team lunch with HCP", "Coffee meeting", "Dinner with HCP", "Working lunch", "Snacks for meeting"],
   870→        "Lodging": ["Hotel stay — field trip", "Lodging — outstation visit", "Guest house — 2 nights"],
   871→        "Samples": ["Sample packs for HCP", "Product samples drop", "Demo kit refill"],
   872→        "Conference": ["Conference registration", "CME event fees", "Doctor meet sponsorship"],
   873→        "Mobile": ["Mobile bill", "Internet recharge", "Roaming charges"],
   874→        "Other": ["Stationery", "Printing materials", "Misc expense"],
   875→    }
   876→
   877→    expense_counter: Dict[date, int] = {}
   878→
   879→    for r in eligible_reps:
   880→        hire = date.fromisoformat(r["hire_date"])
   881→        exit_dt = date.fromisoformat(r["exit_date"]) if r["exit_date"] else END_DATE
   882→        # Iterate weeks
   883→        cur = hire
   884→        while cur <= exit_dt and cur <= END_DATE:
   885→            n_entries = rng.randint(1, 2)
   886→            for _ in range(n_entries):
   887→                # Pick a date in this week
   888→                offset = rng.randint(0, 6)
   889→                ed = cur + timedelta(days=offset)
   890→                if ed > exit_dt or ed > END_DATE or ed < START_DATE:
   891→                    continue
   892→                cat = rng.choices(list(cat_weights.keys()), weights=list(cat_weights.values()), k=1)[0]
   893→                lo, hi = cat_amounts[cat]
   894→                amount = rng.randint(lo, hi)
   895→                reimbursed = rng.random() < 0.90
   896→                compliant = rng.random() < 0.95
   897→                # If non-compliant, slightly increase amount for "policy max exceeded"
   898→                if not compliant and cat == "Lodging":
   899→                    amount = rng.randint(2500, 5000)
   900→                notes = rng.choice(notes_templates[cat])
   901→
   902→                expense_counter[ed] = expense_counter.get(ed, 0) + 1
   903→                expense_id = f"EXP-{ed.strftime('%Y%m%d')}-{expense_counter[ed]:03d}"
   904→                expenses.append({
   905→                    "expense_id": expense_id,
   906→                    "rep_id": r["rep_id"],
   907→                    "expense_date": iso(ed),
   908→                    "category": cat,
   909→                    "amount_inr": amount,
   910→                    "reimbursed": "true" if reimbursed else "false",
   911→                    "policy_compliant": "true" if compliant else "false",
   912→                    "notes": notes,
   913→                })
   914→            cur += timedelta(days=7)
   915→
   916→    return expenses

   919→# --------------------------------------------------------------------------- #
   920→# 7. TARGETS                                                                   #
   921→# --------------------------------------------------------------------------- #
   922→def gen_targets(reps: List[Dict], products: List[Dict], sales: List[Dict]) -> List[Dict]:
   923→    """Generate ~3,600 quarterly targets per rep per product.
   924→
   925→    target_qty = last_year_actual_qty * 1.15 * rep.target_stretch_pct
   926→    For new products, base on therapeutic-area average.
   927→    Each rep covers ~6 products (therapy focus by zone specialty).
   928→    """
   929→    rng = random.Random(SEED + 5)
   930→
   931→    # For each rep, pick ~6 products they cover based on their zone therapy focus
   932→    # Zone -> preferred therapy areas (bias)
   933→    zone_ta_focus = {
   934→        "North": ["Cardio", "Diabetes", "Respiratory"],
   935→        "South": ["Cardio", "Diabetes", "GI"],
   936→        "East": ["Cardio", "Respiratory"],
   937→        "West": ["Diabetes", "GI", "Cardio"],
   938→    }
   939→
   940→    # Sales index: (rep_id, product_id, fy) -> total_qty, total_value
   941→    sales_index: Dict[Tuple[str, str, int], Tuple[int, int]] = {}
   942→    for s in sales:
   943→        d = date.fromisoformat(s["invoice_date"])
   944→        fy = fy_for(d)
   945→        key = (s["rep_id"], s["product_id"], fy)
   946→        cur = sales_index.get(key, (0, 0))
   947→        sales_index[key] = (cur[0] + s["qty_packs"], cur[1] + s["net_value_inr"])
   948→
   949→    # TA average per FY (for new products / reps with no history)
   950→    ta_avg_per_fy: Dict[Tuple[str, int], Tuple[float, float]] = {}
   951→    ta_totals: Dict[Tuple[str, int], Tuple[int, int, int]] = {}
   952→    for (rep_id, pid, fy), (qty, val) in sales_index.items():
   953→        ta = next(p["therapy_area"] for p in products if p["product_id"] == pid)
   954→        k = (ta, fy)
   955→        c = ta_totals.get(k, (0, 0, 0))
   956→        ta_totals[k] = (c[0] + qty, c[1] + val, c[2] + 1)
   957→    for k, (q, v, n) in ta_totals.items():
   958→        ta_avg_per_fy[k] = (q / max(n, 1), v / max(n, 1))
   959→
   960→    # Eligible reps: Field_Reps and Area_Managers (not RM/NH — they have aggregate targets)
   961→    eligible_reps = [r for r in reps if r["role"] in ("Field_Rep", "Area_Manager")]
   962→    targets: List[Dict] = []
   963→
   964→    # Quarter -> fy mapping
   965→    quarter_dates = {
   966→        (2024, "Q1"): (date(2024, 4, 1), date(2024, 6, 30)),
   967→        (2024, "Q2"): (date(2024, 7, 1), date(2024, 9, 30)),
   968→        (2024, "Q3"): (date(2024, 10, 1), date(2024, 12, 31)),
   969→        (2024, "Q4"): (date(2025, 1, 1), date(2025, 3, 31)),
   970→        (2025, "Q1"): (date(2025, 4, 1), date(2025, 6, 30)),
   971→        (2025, "Q2"): (date(2025, 7, 1), date(2025, 9, 30)),
   972→        (2025, "Q3"): (date(2025, 10, 1), date(2025, 12, 31)),
   973→        (2025, "Q4"): (date(2026, 1, 1), date(2026, 3, 31)),
   974→        (2026, "Q1"): (date(2026, 4, 1), date(2026, 6, 30)),
   975→    }
   976→    # We'll generate targets for FY24, FY25, FY26 (Q1-Q4 each)
   977→    fys = [2024, 2025, 2026]
   978→    quarters = ["Q1", "Q2", "Q3", "Q4"]
   979→
   980→    for r in eligible_reps:
   981→        # Pick ~6 products for this rep
   982→        zone = r["zone"]
   983→        preferred_tas = zone_ta_focus[zone]
   984→        # 4 products from preferred TAs + 2 from others
   985→        preferred_products = [p for p in products if p["therapy_area"] in preferred_tas]
   986→        other_products = [p for p in products if p["therapy_area"] not in preferred_tas]
   987→        rng.shuffle(preferred_products)
   988→        rng.shuffle(other_products)
   989→        rep_products = preferred_products[:4] + other_products[:2]
   990→
   991→        for fy in fys:
   992→            for q in quarters:
   993→                # Only if rep was active for any part of this quarter
   994→                q_start, q_end = quarter_dates.get((fy, q), (None, None))
   995→                if q_start is None:
   996→                    continue
   997→                hire = date.fromisoformat(r["hire_date"])
   998→                exit_dt = date.fromisoformat(r["exit_date"]) if r["exit_date"] else END_DATE
   999→                # Skip if rep hired after quarter end OR exited before quarter start
  1000→                if hire > q_end or exit_dt < q_start:
  1001→                    continue
  1002→                for p in rep_products:
  1003→                    # Look up last year actual
  1004→                    prev_fy = fy - 1
  1005→                    prev_actual_qty, prev_actual_val = sales_index.get((r["rep_id"], p["product_id"], prev_fy), (0, 0))
  1006→                    if prev_actual_qty == 0:
  1007→                        # New product or new rep: use TA average for prior FY
  1008→                        ta_avg_qty, ta_avg_val = ta_avg_per_fy.get((p["therapy_area"], prev_fy), (40, 20000))
  1009→                        base_qty = int(ta_avg_qty / 4)  # quarterly
  1010→                        base_val = int(ta_avg_val / 4)
  1011→                    else:
  1012→                        base_qty = prev_actual_qty / 4  # convert annual to quarterly
  1013→                        base_val = prev_actual_val / 4
  1014→                    stretch = float(r["target_stretch_pct"])
  1015→                    growth = 1.15
  1016→                    target_qty = max(10, int(base_qty * growth * stretch))
  1017→                    target_value = max(1000, int(base_val * growth * stretch))
  1018→                    # ID: TGT-YYYYQn-REPxxx-PRDCARD01 (compact form)
  1019→                    rep_compact = r["rep_id"].replace("REP-", "REP")
  1020→                    prd_compact = p["product_id"].replace("PRD-", "").replace("-", "")
  1021→                    target_id = f"TGT-{fy}{q}-{rep_compact}-{prd_compact}"
  1022→                    targets.append({
  1023→                        "target_id": target_id,
  1024→                        "rep_id": r["rep_id"],
  1025→                        "product_id": p["product_id"],
  1026→                        "fy": fy,
  1027→                        "quarter": q,
  1028→                        "target_qty": target_qty,
  1029→                        "target_value_inr": target_value,
  1030→                    })
  1031→
  1032→    return targets
